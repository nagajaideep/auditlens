SELECT type,
       COUNT(*)                              AS n_tx,
       ROUND(SUM(amount)::numeric, 0)        AS total_amount,
       ROUND(AVG(amount)::numeric, 2)        AS avg_amount,
       SUM(isfraud)                          AS n_fraud,
       ROUND(100.0 * AVG(isfraud), 4)        AS fraud_pct
FROM transactions_raw
GROUP BY type
ORDER BY n_fraud DESC;
